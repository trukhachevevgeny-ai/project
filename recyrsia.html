Программа Рекурсия. Функция вызывает саму себя

Рекурсия — это когда функция внутри себя вызывает саму себя с новыми данными. Каждый такой вызов «уходит в глубину». Перетащите кружок в серую зону 3 раза, чтобы достичь базового случая — и рекурсия завершится.

program Recur;

{ Рекурсивный отсчёт: печатает n, n-1, ..., 1 }

procedure Count(n : integer);
begin
  if n = 0 then
    exit;              { базовый случай: остановка }

  writeln(n);            { печатаем текущее число }
  Count(n - 1);          { рекурсивный вызов }
end;

begin
  Count(3);               { запуск: 3, 2, 1 }
end.
# Рекурсивный отсчёт: печатает n, n-1, ..., 1

def count(n):
    if n == 0:
        return              # базовый случай

    print(n)                # печатаем число
    count(n - 1)             # рекурсивный вызов

count(3)                    # запуск: 3, 2, 1
// Рекурсивный отсчёт: печатает n, n-1, ..., 1
public class Recur {
    static void count(int n) {
        if (n == 0) {
            return;              // базовый случай
        }

        System.out.println(n);    // печатаем число
        count(n - 1);         // рекурсивный вызов
    }

    public static void main(String[] args) {
        count(3);                // запуск: 3, 2, 1
    }
}
