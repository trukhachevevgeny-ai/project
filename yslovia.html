Программа условие. Сравнение цветов

Справа - условие, слева - то, что сравнивают.
Если цвета совпадут — центральный круг станет белым, иначе останется чёрным. 

program Uslovie;

  иначе вывести 0 }

var
  x : integer;  { вход }

begin
  readln(x);           { читаем число }

  if x = 1 then
    writeln(1)         { если 1 — выводим 1 }
  else
    writeln(0);        { иначе — выводим 0 }
end.

// иначе вывести 0
import java.util.Scanner;

public class Uslovie {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int x = sc.nextInt();      // читаем число

        if (x == 1) {
            System.out.println(1);   // если 1 — выводим 1
        } else {
            System.out.println(0);   // иначе — выводим 0
        }

        sc.close();
    }
}
