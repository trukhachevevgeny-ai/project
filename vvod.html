// Программа: что на входе — то и на выходе
import java.util.Scanner;

public class Program {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        String c = sc.nextLine();   // читаем вход
        System.out.println(c);         // выводим то же самое
        sc.close();
    }
}
